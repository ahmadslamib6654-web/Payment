import os import logging import asyncio from flask import Flask, request from telegram import Update from telegram.ext import ( Application, CommandHandler, ContextTypes ) # إعداد اللوج logging.basicConfig( format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO ) logger = logging.getLogger(__name__) # ========== الإعدادات ========== BOT_TOKEN = os.getenv("BOT_TOKEN", "") # ضع التوكن الجديد في متغير بيئة OWNER_ID = 16420566 # رقم المستخدم (أنت) # إعدادات الويب هوك WEBHOOK_URL = os.getenv("WEBHOOK_URL", "https://yourdomain.com/webhook") PORT = int(os.environ.get("PORT", "8443")) # ========== أوامر البوت ========== async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user if user.id == OWNER_ID: await update.message.reply_text("مرحباً أيها الأدمن ✅") else: await update.message.reply_text("هذا بوت خاص ❌") async def status(update: Update, context: ContextTypes.DEFAULT_TYPE): await update.message.reply_text("البوت شغال ✅") # ========== معالجة الأخطاء ========== async def error_handler(update: object, context: ContextTypes.DEFAULT_TYPE): logger.error("Exception: %s", context.error, exc_info=True) try: if OWNER_ID: await context.bot.send_message( chat_id=OWNER_ID, text=f"⚠️ خطأ في البوت:\n{context.error}" ) except Exception as e: logger.error("Failed to send error to owner: %s", e) # ========== الإعداد ========== app = Flask(__name__) tg_app = Application.builder().token(BOT_TOKEN).build() tg_app.add_handler(CommandHandler("start", start)) tg_app.add_handler(CommandHandler("status", status)) tg_app.add_error_handler(error_handler) @app.route("/webhook", methods=["POST"]) def webhook(): update = Update.de_json(request.get_json(force=True), tg_app.bot) tg_app.update_queue.put_nowait(update) return "ok" # ========== تشغيل ========== if __name__ == "__main__": # إعداد الويب هوك عند التشغيل async def set_webhook(): await tg_app.bot.set_webhook(WEBHOOK_URL) asyncio.run(set_webhook()) tg_app.run_webhook( listen="0.0.0.0", port=PORT, url_path="webhook", webhook_url=WEBHOOK_URL )
