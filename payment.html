# bot.py # متوافق مع python-telegram-bot v20+ import os import asyncio import logging import json from pathlib import Path from telegram import Update from telegram.ext import ( Application, CommandHandler, ContextTypes, PicklePersistence ) # ====== إعدادات ====== # ضع التوكن كمتغير بيئة BOT_TOKEN (لا تضع التوكن داخل الكود) BOT_TOKEN = os.getenv("BOT_TOKEN", "") if not BOT_TOKEN: raise SystemExit("ERROR: BOT_TOKEN environment variable is required. Generate a new token from BotFather and export BOT_TOKEN.") # Admin (Owner) ID — المستخدم الذي أعطيته OWNER_ID = 16420566 # ملف حفظ الحالة (admins, banned, etc) DATA_FILE = Path("bot_data.json") # مهيئ السجل logging.basicConfig( format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO ) logger = logging.getLogger(__name__) # ====== دوال المساعدة للعمل مع ملف البيانات ====== def load_data(): if DATA_FILE.exists(): with DATA_FILE.open("r", encoding="utf-8") as f: return json.load(f) # بنية البداية return {"admins": [OWNER_ID], "banned": []} def save_data(data): with DATA_FILE.open("w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2) # ====== handlers ====== async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user data = load_data() if user.id in data.get("banned", []): await update.message.reply_text("حسابك محظور من استخدام هذا البوت.") return if user.id in data.get("admins", []): await update.message.reply_text("أهلاً أيها الأدمن ✅\nاستخدم /help لرؤية الأوامر.") else: await update.message.reply_text("أهلاً بك! هذا بوت خاص — لا تملك صلاحيات إدارية.") async def help_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE): data = load_data() if update.effective_user.id not in data.get("admins", []): await update.message.reply_text("عذراً، هذه الأوامر مخصصة للمشرفين فقط.") return txt = ( "قائمة الأوامر الإدارية:\n\n" "/start - بدء\n" "/help - هذه الرسالة\n" "/status - حالة البوت\n" "/admins - قائمة الأدمن\n" "/add_admin - إضافة أدمن\n" "/remove_admin - إزالة أدمن\n" "/ban - حظر مستخدم\n" "/unban - فك الحظر\n" "/broadcast - بث رسالة للمستخدمين (حاليًا يرسل فقط للأدمنز)\n" "/shutdown - إيقاف البوت (للـ owner فقط)\n" ) await update.message.reply_text(txt) async def status(update: Update, context: ContextTypes.DEFAULT_TYPE): data = load_data() await update.message.reply_text(f"البوت يعمل ✅\nعدد الأدمن: {len(data.get('admins',[]))}") async def admins_list(update: Update, context: ContextTypes.DEFAULT_TYPE): data = load_data() admins = data.get("admins", []) await update.message.reply_text("قائمة الأدمن:\n" + "\n".join(map(str, admins))) def is_owner(user_id: int): return user_id == OWNER_ID async def add_admin(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user data = load_data() if user.id not in data.get("admins", []): await update.message.reply_text("فقط الأدمنز يمكنهم استخدام هذا الأمر.") return if not context.args: await update.message.reply_text("استخدام: /add_admin ") return try: new_id = int(context.args[0]) except ValueError: await update.message.reply_text("أدخل رقم معرّف صحيح.") return if new_id in data["admins"]: await update.message.reply_text("هذا المستخدم موجود بالفعل كأدمن.") return data["admins"].append(new_id) save_data(data) await update.message.reply_text(f"تمت إضافة الأدمن: {new_id}") async def remove_admin(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user data = load_data() if user.id not in data.get("admins", []): await update.message.reply_text("فقط الأدمنز يمكنهم استخدام هذا الأمر.") return if not context.args: await update.message.reply_text("استخدام: /remove_admin ") return try: rid = int(context.args[0]) except ValueError: await update.message.reply_text("أدخل رقم معرّف صحيح.") return if rid == OWNER_ID: await update.message.reply_text("لا يمكنك إزالة الـ owner.") return if rid not in data["admins"]: await update.message.reply_text("هذا المستخدم ليس أدمن.") return data["admins"].remove(rid) save_data(data) await update.message.reply_text(f"تمت إزالة الأدمن: {rid}") async def ban_user(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user data = load_data() if user.id not in data.get("admins", []): await update.message.reply_text("فقط الأدمنز يمكنهم استخدام هذا الأمر.") return if not context.args: await update.message.reply_text("استخدام: /ban ") return try: uid = int(context.args[0]) except ValueError: await update.message.reply_text("أدخل رقم معرّف صحيح.") return if uid in data.get("banned", []): await update.message.reply_text("الحساب محظور بالفعل.") return data.setdefault("banned", []).append(uid) save_data(data) await update.message.reply_text(f"تم حظر: {uid}") async def unban_user(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user data = load_data() if user.id not in data.get("admins", []): await update.message.reply_text("فقط الأدمنز يمكنهم استخدام هذا الأمر.") return if not context.args: await update.message.reply_text("استخدام: /unban ") return try: uid = int(context.args[0]) except ValueError: await update.message.reply_text("أدخل رقم معرّف صحيح.") return if uid not in data.get("banned", []): await update.message.reply_text("هذا الحساب غير محظور.") return data["banned"].remove(uid) save_data(data) await update.message.reply_text(f"تم فك الحظر عن: {uid}") async def broadcast(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user data = load_data() if user.id not in data.get("admins", []): await update.message.reply_text("فقط الأدمنز يمكنهم استخدام هذا الأمر.") return if not context.args: await update.message.reply_text("استخدام: /broadcast ") return message = " ".join(context.args) # مثال بسيط: نرسل الرسالة لكل الأدمنز فقط (تعديل حسب حاجتك) targets = data.get("admins", []) sent = 0 for uid in targets: try: await context.bot.send_message(chat_id=uid, text=f"[Broadcast]\n{message}") sent += 1 except Exception as e: logger.warning("Failed to send to %s: %s", uid, e) await update.message.reply_text(f"أرسلت الرسالة إلى {sent} حساب/حسابات.") async def shutdown(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user if not is_owner(user.id): await update.message.reply_text("فقط الـ owner يمكنه إيقاف البوت.") return await update.message.reply_text("جاري إيقاف البوت... ✅") # graceful stop asyncio.get_event_loop().stop() # ====== main ====== def main(): data = load_data() # حفظ الملف إن لم يكن موجودًا save_data(data) app = Application.builder().token(BOT_TOKEN).build() # أوامر app.add_handler(CommandHandler("start", start)) app.add_handler(CommandHandler("help", help_cmd)) app.add_handler(CommandHandler("status", status)) app.add_handler(CommandHandler("admins", admins_list)) app.add_handler(CommandHandler("add_admin", add_admin)) app.add_handler(CommandHandler("remove_admin", remove_admin)) app.add_handler(CommandHandler("ban", ban_user)) app.add_handler(CommandHandler("unban", unban_user)) app.add_handler(CommandHandler("broadcast", broadcast)) app.add_handler(CommandHandler("shutdown", shutdown)) logger.info("Starting bot...") app.run_polling() if __name__ == "__main__": main()
